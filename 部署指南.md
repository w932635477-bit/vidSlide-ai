### 第3步：连接 VideoProcessingService

当前 `WorkflowOrchestrator` 使用模拟数据。要连接实际服务：

#### 1. 修改 WorkflowOrchestrator.js

```javascript
// vidslide-ai/src/services/WorkflowOrchestrator.js

// 导入实际服务
import { VideoProcessingService } from './VideoProcessingService.js'
import { getBaiduNLPService } from './BaiduNLPService.js'

export class WorkflowOrchestrator {
  constructor() {
    // 添加服务实例
    this.videoService = new VideoProcessingService()
    this.nlpService = getBaiduNLPService()

    // ... 其他代码
  }

  /**
   * 步骤1: 分析视频（使用实际服务）
   */
  async analyzeVideo(inputData, log) {
    const { videoFile } = inputData

    log('加载视频文件')

    // 使用实际的 VideoProcessingService
    const metadata = await this.videoService.loadVideo(videoFile)

    log('提取关键帧')
    const keyframes = await this.videoService.extractKeyframes({
      interval: 2,
      maxFrames: 50,
      onProgress: (p) => log(`提取关键帧: ${p}%`)
    })

    log('检测场景')
    const scenes = await this.videoService.detectScenes(keyframes)

    log('语音识别')
    const transcript = await this.videoService.recognizeSpeech((progress) => {
      log(`语音识别: ${progress.progress}%`)
    })

    log('提取关键词')
    const keywords = transcript.keywords || []

    log('分析内容类型')
    const contentType = this.detectContentType(transcript.text, keywords)

    return {
      ...inputData,
      metadata,
      keyframes,
      scenes,
      transcript: transcript.text,
      keywords,
      contentType
    }
  }
}
```

### 第4步：集成 MaterialService

#### 1. 修改素材匹配步骤

```javascript
// vidslide-ai/src/services/WorkflowOrchestrator.js

import { MaterialService } from './MaterialService.js'

export class WorkflowOrchestrator {
  constructor() {
    this.materialService = new MaterialService()
    // ... 其他代码
  }

  /**
   * 步骤3: 匹配素材（使用实际服务）
   */
  async matchMaterials(inputData, log) {
    const { keywords } = inputData
    const materials = []

    log('开始搜索素材')

    for (let i = 0; i < Math.min(keywords.length, 5); i++) {
      const keyword = keywords[i]
      log(`搜索素材: ${keyword.text}`)

      try {
        // 使用实际的 MaterialService
        const results = await this.materialService.searchMaterials({
          keyword: keyword.text,
          platforms: ['unsplash', 'pexels'],
          maxResults: 3
        })

        if (results && results.length > 0) {
          materials.push({
            keyword: keyword.text,
            material: results[0] // 选择第一个结果
          })
          log(`找到 ${results.length} 个素材`)
        } else {
          log(`未找到素材: ${keyword.text}`)
        }
      } catch (error) {
        log(`素材搜索失败: ${error.message}`)
        // 继续处理其他关键词
      }
    }

    log(`总共找到 ${materials.length} 个素材`)

    return {
      ...inputData,
      materials
    }
  }
}
```

### 第6步：性能监控

#### 1. 添加性能监控

```javascript
// vidslide-ai/src/utils/performanceMonitor.js

export class PerformanceMonitor {
  constructor(config) {
    this.config = config
    this.metrics = []
  }

  /**
   * 记录工作流执行
   */
  trackWorkflow(workflowId, data) {
    const metric = {
      type: 'workflow_execution',
      workflowId,
      duration: data.duration,
      success: data.success,
      steps: data.steps,
      timestamp: Date.now()
    }

    this.metrics.push(metric)

    // 发送到分析服务
    if (this.config.enableAnalytics) {
      this.sendMetric(metric)
    }
  }

  /**
   * 记录错误
   */
  trackError(error, context) {
    const errorData = {
      type: 'error',
      message: error.message,
      context,
      timestamp: Date.now()
    }

    // 发送错误报告
    if (this.config.errorReporting.enabled) {
      this.sendError(errorData)
    }
  }

  /**
   * 发送指标
   */
  async sendMetric(metric) {
    // 采样
    if (Math.random() > this.config.sampleRate) {
      return
    }

    try {
      await fetch('/api/metrics', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(metric)
      })
    } catch (error) {
      console.error('发送指标失败:', error)
    }
  }

  /**
   * 发送错误
   */
  async sendError(errorData) {
    try {
      await fetch(this.config.errorReporting.endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(errorData)
      })
    } catch (error) {
      console.error('发送错误报告失败:', error)
    }
  }

  /**
   * 获取统计信息
   */
  getStatistics() {
    const total = this.metrics.length
    const successful = this.metrics.filter(m => m.success).length
    const failed = total - successful

    const durations = this.metrics.map(m => m.duration)
    const avgDuration = durations.reduce((a, b) => a + b, 0) / total

    return {
      total,
      successful,
      failed,
      successRate: (successful / total) * 100,
      avgDuration
    }
  }
}

// 导出单例
export const performanceMonitor = new PerformanceMonitor({
  enableAnalytics: import.meta.env.PROD,
  sampleRate: 0.1,
  errorReporting: {
    enabled: import.meta.env.PROD,
    endpoint: '/api/errors'
  }
})
```

#### 2. 集成到 Agent

```javascript
// vidslide-ai/src/services/MasterAutoGenerationAgentV2.js

import { performanceMonitor } from '@/utils/performanceMonitor.js'

export class MasterAutoGenerationAgent {
  async autoGenerate(videoFile, onProgress) {
    const startTime = Date.now()
    const workflowId = `workflow-${Date.now()}`

    try {
      const result = await this.orchestrator.executeWorkflow(videoFile)

      // 记录成功
      performanceMonitor.trackWorkflow(workflowId, {
        duration: Date.now() - startTime,
        success: true,
        steps: this.getAllSteps().length
      })

      return result

    } catch (error) {
      // 记录失败
      performanceMonitor.trackWorkflow(workflowId, {
        duration: Date.now() - startTime,
        success: false,
        steps: this.getAllSteps().length
      })

      // 记录错误
      performanceMonitor.trackError(error, {
        workflowId,
        videoFile: videoFile.name
      })

      throw error
    }
  }
}
```

### 第7步：告警配置

```javascript
// vidslide-ai/src/utils/alerting.js

export class AlertingService {
  constructor() {
    this.thresholds = {
      executionTime: 60000, // 60秒
      errorRate: 0.1, // 10%
      memoryUsage: 200 * 1024 * 1024 // 200MB
    }
  }

  /**
   * 检查并发送告警
   */
  checkAndAlert(stats) {
    // 检查执行时间
    if (stats.avgDuration > this.thresholds.executionTime) {
      this.sendAlert({
        type: 'performance',
        severity: 'warning',
        message: `工作流平均执行时间过长: ${stats.avgDuration}ms`,
        data: stats
      })
    }

    // 检查错误率
    const errorRate = stats.failed / stats.total
    if (errorRate > this.thresholds.errorRate) {
      this.sendAlert({
        type: 'reliability',
        severity: 'critical',
        message: `工作流错误率过高: ${(errorRate * 100).toFixed(2)}%`,
        data: stats
      })
    }
  }

  /**
   * 发送告警
   */
  async sendAlert(alert) {
    console.error('🚨 告警:', alert)

    // 发送到告警服务
    try {
      await fetch('/api/alerts', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(alert)
      })
    } catch (error) {
      console.error('发送告警失败:', error)
    }
  }
}

export const alerting = new AlertingService()
```

### 4. 配置 Nginx

```nginx
server {
    listen 80;
    server_name vidslide.ai;

    root /var/www/vidslide-ai;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # API 代理
    location /api/ {
        proxy_pass http://localhost:3000/;
    }

    # 性能优化
    gzip on;
    gzip_types text/plain text/css application/json application/javascript;
}
```
