```javascript
// src/services/CLIPMatcher.js
import { pipeline } from '@xenova/transformers'

class CLIPMatcher {
  constructor() {
    this.model = null
    this.isInitialized = false
  }

  async initialize() {
    try {
      console.log('🔄 加载CLIP模型...')

      // 使用轻量级CLIP模型
      this.model = await pipeline('zero-shot-image-classification',
        'Xenova/clip-vit-base-patch32'
      )

      this.isInitialized = true
      console.log('✅ CLIP模型加载成功')
    } catch (error) {
      console.error('❌ CLIP模型加载失败:', error)
      // 降级到模拟实现
      this.isInitialized = false
    }
  }

  async matchTextToImages(text, images) {
    if (!this.isInitialized) {
      // 降级到关键词匹配
      return this.fallbackKeywordMatch(text, images)
    }

    try {
      const similarities = []

      for (const image of images) {
        const result = await this.model(image.url || image.dataUrl, [text])
        similarities.push(result[0].score)
      }

      return similarities
    } catch (error) {
      console.error('CLIP匹配失败:', error)
      return this.fallbackKeywordMatch(text, images)
    }
  }

  fallbackKeywordMatch(text, images) {
    // 现有的关键词匹配逻辑
    return images.map(() => Math.random() * 0.5 + 0.3)
  }
}

export default new CLIPMatcher()
```

```javascript
class CLIPMatcher {
  constructor() {
    this.apiEndpoint = process.env.VITE_CLIP_API_URL || 'http://localhost:8000/clip'
  }

  async matchTextToImages(text, images) {
    try {
      const response = await fetch(this.apiEndpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          text,
          images: images.map(img => img.url || img.dataUrl)
        })
      })

      const result = await response.json()
      return result.similarities
    } catch (error) {
      console.error('CLIP API调用失败:', error)
      return this.fallbackKeywordMatch(text, images)
    }
  }
}
```

**后端实现（Python + FastAPI）**:
```python
from fastapi import FastAPI
from transformers import CLIPProcessor, CLIPModel
import torch
from PIL import Image
import requests
from io import BytesIO

app = FastAPI()
model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

@app.post("/clip")
async def match_text_to_images(data: dict):
    text = data["text"]
    image_urls = data["images"]

    images = []
    for url in image_urls:
        if url.startswith('data:'):
            # 处理base64图片
            pass
        else:
            response = requests.get(url)
            images.append(Image.open(BytesIO(response.content)))

    inputs = processor(
        text=[text] * len(images),
        images=images,
        return_tensors="pt",
        padding=True
    )

    with torch.no_grad():
        outputs = model(**inputs)
        similarities = outputs.logits_per_text[0].softmax(dim=0).tolist()

    return {"similarities": similarities}
```

**新增监控服务**:

```javascript
// src/services/MatchQualityMonitor.js
class MatchQualityMonitor {
  constructor() {
    this.stats = {
      totalSearches: 0,
      successfulMatches: 0,
      failedMatches: 0,
      avgMatchRate: 0,
      matchTypeDistribution: {
        exact: 0,
        semantic: 0,
        industry: 0,
        keyword: 0
      },
      authorizationTriggers: 0,
      authorizationGranted: 0,
      authorizationDenied: 0,
      localOnlyMode: 0
    }

    this.searchHistory = []
    this.maxHistorySize = 100
  }

  /**
   * 记录搜索结果
   */
  recordSearch(searchData) {
    this.stats.totalSearches++

    const {
      keywords,
      matchRate,
      matchedCount,
      matchType,
      authorizationTriggered,
      authorizationResult
    } = searchData

    // 记录匹配成功/失败
    if (matchRate >= 0.8) {
      this.stats.successfulMatches++
    } else {
      this.stats.failedMatches++
    }

    // 更新平均匹配率
    this.stats.avgMatchRate =
      (this.stats.avgMatchRate * (this.stats.totalSearches - 1) + matchRate) /
      this.stats.totalSearches

    // 记录匹配类型分布
    if (matchType) {
      this.stats.matchTypeDistribution[matchType]++
    }

    // 记录授权情况
    if (authorizationTriggered) {
      this.stats.authorizationTriggers++

      if (authorizationResult === 'granted') {
        this.stats.authorizationGranted++
      } else if (authorizationResult === 'denied') {
        this.stats.authorizationDenied++
      } else if (authorizationResult === 'local_only') {
        this.stats.localOnlyMode++
      }
    }

    // 添加到历史记录
    this.searchHistory.push({
      timestamp: Date.now(),
      keywords,
      matchRate,
      matchedCount,
      matchType,
      authorizationTriggered,
      authorizationResult
    })

    // 限制历史记录大小
    if (this.searchHistory.length > this.maxHistorySize) {
      this.searchHistory.shift()
    }

    // 持久化到localStorage
    this.saveToStorage()
  }

  /**
   * 获取统计数据
   */
  getStats() {
    return {
      ...this.stats,
      successRate: this.stats.totalSearches > 0
        ? this.stats.successfulMatches / this.stats.totalSearches
        : 0,
      authorizationRate: this.stats.totalSearches > 0
        ? this.stats.authorizationTriggers / this.stats.totalSearches
        : 0,
      authorizationGrantRate: this.stats.authorizationTriggers > 0
        ? this.stats.authorizationGranted / this.stats.authorizationTriggers
        : 0
    }
  }

  /**
   * 获取失败原因分析
   */
  analyzeFailures() {
    const failures = this.searchHistory.filter(s => s.matchRate < 0.8)

    const analysis = {
      totalFailures: failures.length,
      commonKeywords: this.getCommonKeywords(failures),
      avgMatchRate: failures.reduce((sum, f) => sum + f.matchRate, 0) / failures.length,
      recommendations: []
    }

    // 生成优化建议
    if (analysis.avgMatchRate < 0.5) {
      analysis.recommendations.push('本地素材库覆盖不足，建议扩充素材库')
    }

    if (this.stats.authorizationDenied > this.stats.authorizationGranted) {
      analysis.recommendations.push('用户倾向拒绝外部素材，建议优化本地素材质量')
    }

    return analysis
  }

  /**
   * 获取常见失败关键词
   */
  getCommonKeywords(failures) {
    const keywordCount = {}

    failures.forEach(f => {
      f.keywords.forEach(kw => {
        keywordCount[kw] = (keywordCount[kw] || 0) + 1
      })
    })

    return Object.entries(keywordCount)
      .sort((a, b) => b[1] - a[1])
      .slice(0, 10)
      .map(([keyword, count]) => ({ keyword, count }))
  }

  /**
   * 持久化到localStorage
   */
  saveToStorage() {
    try {
      localStorage.setItem('match_quality_stats', JSON.stringify(this.stats))
      localStorage.setItem('match_quality_history', JSON.stringify(this.searchHistory))
    } catch (error) {
      console.warn('保存监控数据失败:', error)
    }
  }

  /**
   * 从localStorage加载
   */
  loadFromStorage() {
    try {
      const stats = localStorage.getItem('match_quality_stats')
      const history = localStorage.getItem('match_quality_history')

      if (stats) {
        this.stats = JSON.parse(stats)
      }

      if (history) {
        this.searchHistory = JSON.parse(history)
      }
    } catch (error) {
      console.warn('加载监控数据失败:', error)
    }
  }

  /**
   * 重置统计数据
   */
  reset() {
    this.stats = {
      totalSearches: 0,
      successfulMatches: 0,
      failedMatches: 0,
      avgMatchRate: 0,
      matchTypeDistribution: {
        exact: 0,
        semantic: 0,
        industry: 0,
        keyword: 0
      },
      authorizationTriggers: 0,
      authorizationGranted: 0,
      authorizationDenied: 0,
      localOnlyMode: 0
    }

    this.searchHistory = []
    this.saveToStorage()
  }
}

export default new MatchQualityMonitor()
```

```javascript
class LocalMaterialLibrary {
  constructor() {
    this.searchIndex = null
  }

  async buildSearchIndex() {
    // 使用Lunr.js或Fuse.js构建搜索索引
    const Fuse = await import('fuse.js')

    this.searchIndex = new Fuse.default(this.materials, {
      keys: ['name', 'tags', 'keywords', 'description'],
      threshold: 0.3,
      includeScore: true
    })
  }

  searchMaterials(query) {
    if (!this.searchIndex) {
      return this.fallbackSearch(query)
    }

    const results = this.searchIndex.search(query)
    return results.map(r => ({
      ...r.item,
      relevanceScore: 1 - r.score
    }))
  }
}
```

#### 3.2 缓存优化

```javascript
// src/services/MaterialService.js
class MaterialService {
  constructor() {
    this.searchCache = new Map()
    this.cacheMaxSize = 50
    this.cacheTTL = 5 * 60 * 1000 // 5分钟
  }

  async searchMaterials(query, options = {}) {
    const cacheKey = this.generateCacheKey(query, options)
    const cached = this.searchCache.get(cacheKey)

    if (cached && Date.now() - cached.timestamp < this.cacheTTL) {
      console.log('💾 使用缓存结果')
      return cached.data
    }

    const results = await this.performSearch(query, options)

    // 添加到缓存
    this.searchCache.set(cacheKey, {
      data: results,
      timestamp: Date.now()
    })

    // 限制缓存大小
    if (this.searchCache.size > this.cacheMaxSize) {
      const firstKey = this.searchCache.keys().next().value
      this.searchCache.delete(firstKey)
    }

    return results
  }
}
```
