// 智能素材服务 - 集成多个专业平台
class IntelligentMaterialService {
  constructor() {
    this.platforms = {
      unsplash: new UnsplashAPI(),
      pexels: new PexelsAPI(),
      pixabay: new PixabayAPI()
    };
    this.cache = new IndexedDBCache();
  }

  // 智能搜索 - 根据关键词选择最佳平台
  async searchMaterials(keyword, options = {}) {
    const analysis = await this.analyzeKeyword(keyword);

    // 智能调度：选择最合适的平台
    const platform = this.selectBestPlatform(analysis, options);

    // 并行搜索多个平台提高效率
    const results = await this.parallelSearch(keyword, platform);

    // 质量筛选和去重
    const filtered = await this.filterQuality(results);

    // 本地缓存
    await this.cache.store(filtered);

    return filtered;
  }

  // 分析关键词特征
  async analyzeKeyword(keyword) {
    return {
      language: this.detectLanguage(keyword),
      category: this.categorizeKeyword(keyword),
      quality: this.assessQuality(keyword),
      platforms: this.recommendPlatforms(keyword)
    };
  }

  // 选择最佳平台
  selectBestPlatform(analysis, options) {
    // 根据关键词特征选择平台
    if (analysis.category === 'photography') {
      return ['unsplash', 'pexels'];
    }
    if (analysis.category === 'illustrations') {
      return ['pixabay', 'unsplash'];
    }
    return ['unsplash', 'pexels', 'pixabay'];
  }
}
💾 IndexedDB缓存系统
// 本地素材缓存 - 确保离线可用
class IndexedDBCache {
  constructor() {
    this.dbName = 'VidSlideMaterials';
    this.version = 1;
  }

  async init() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, this.version);

      request.onerror = () => reject(request.error);
      request.onsuccess = () => resolve(request.result);

      request.onupgradeneeded = (event) => {
        const db = event.target.result;

        // 素材存储
        if (!db.objectStoreNames.contains('materials')) {
          const store = db.createObjectStore('materials', { keyPath: 'id' });
          store.createIndex('keyword', 'keyword', { unique: false });
          store.createIndex('platform', 'platform', { unique: false });
          store.createIndex('quality', 'quality', { unique: false });
        }

        // 搜索历史
        if (!db.objectStoreNames.contains('searchHistory')) {
          db.createObjectStore('searchHistory', { keyPath: 'id' });
        }
      };
    });
  }

  // 存储素材
  async store(materials) {
    const db = await this.init();
    const transaction = db.transaction(['materials'], 'readwrite');
    const store = transaction.objectStore('materials');

    for (const material of materials) {
      await new Promise((resolve, reject) => {
        const request = store.put(material);
        request.onsuccess = () => resolve();
        request.onerror = () => reject(request.error);
      });
    }
  }
}
