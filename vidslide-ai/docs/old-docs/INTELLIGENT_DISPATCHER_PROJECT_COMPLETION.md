```javascript
// MaterialService深度集成
class MaterialService {
  constructor() {
    this.intelligentDispatcher = IntelligentDispatcher
  }

  async searchMaterials(query) {
    // 智能调度决策
    const decision = await this.intelligentDispatcher.dispatch(query)

    // 基于决策执行搜索

}
```
