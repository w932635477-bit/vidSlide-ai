## 🎯 项目概述

### 背景

基于理想效果图片的需求，当前项目流程无法实现复杂的多层视频效果。需要构建一个**多智能体蜂群系统**，通过智能体协作完成：

1. **三类画面系统**
   - 原视频画面（过渡用）
   - 原视频+卡片画面（观点表述）
   - 多层组合画面（5层结构，深度解释）

2. **智能协作**
   - 内容理解（语音识别 + 文心一言分析）
   - 场景拆解（智能规划时间轴）
   - 素材生成（豆包生成 + 多重验证）
   - 质量控制（多维度检查）
   - 视频合成（5层画面合成）

### 核心挑战

1. **准确性要求高**
   - 素材内容必须准确（90%+）
   - 场景拆解必须合理（95%+）
   - 质量控制必须严格（98%+）

2. **协作复杂度高**
   - 5个核心智能体
   - 多层依赖关系
   - 错误恢复机制
   - 质量门禁系统

3. **一人开发**
   - 代码量需控制
   - 架构需清晰
   - 可维护性要强

## ⏱️ 时间轴系统设计

### 核心理念

时间轴是整个视频生成系统的**基础设施**，它必须：
1. **优先于所有智能体工作**：在内容分析之前就建立
2. **基于视频实际结构**：使用精确的语音时间戳，而非"大致时间"
3. **与UI完全吻合**：后端生成的时间轴可以直接在前端UI中展示和编辑
4. **支持自然插入**：在语音停顿处插入卡片，不打断说话

### 三层时间轴架构

```
┌─────────────────────────────────────────────────────────┐
│  Layer 1: 基础时间轴（TimelineBuilder）                  │
│  - 视频基础信息（时长、FPS）                              │
│  - 语音分段（带精确时间戳）                               │
│  - 自然断点（停顿、句子结束）                             │
│  - 场景变化点（可选）                                     │
│                                                          │
│  输出：baseTimeline                                      │
└─────────────────────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────────────────────┐
│  Layer 2: 场景时间轴（SceneDesigner）                    │
│  - 基于Layer 1生成场景                                   │
│  - 映射到UI的tracks和clips                               │
│  - 包含所有元数据（卡片文本、素材需求等）                 │
│                                                          │
│  输出：uiTimeline                                        │
└─────────────────────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────────────────────┐
│  Layer 3: UI时间轴（前端TimelineEditor）                 │
│  - 直接使用Layer 2的数据                                 │
│  - 支持用户编辑和调整                                     │
│  - 实时预览                                              │
│  - 拖拽、调整clips                                       │
└─────────────────────────────────────────────────────────┘
```

### 优势总结

1. **精确性**
   - 基于ASR的精确时间戳，不是"大致时间"
   - 词级别时间戳支持更精细的控制

2. **自然性**
   - 在语音停顿处插入卡片，不打断说话
   - 基于实际语音节奏，而非固定时长规则

### 0. TimelineBuilder（时间轴构建器）

#### 核心职责

```javascript
/**
 * TimelineBuilder - 时间轴构建器
 *
 * 职责：
 * 1. 构建基础时间轴（Layer 1）
 * 2. 提供精确的时间戳和插入点
 * 3. 为后续智能体提供时间轴基础
 *
 * 优先级：最高（必须第一个执行）
 */
class TimelineBuilder {
  constructor(options = {}) {
    this.name = 'TimelineBuilder';
    this.asrService = new BaiduASRService();
    this.logger = options.logger || console;
    this.errorHandler = options.errorHandler;
  }

  /**
   * 构建基础时间轴
   * @param {Object} input
   * @param {string} input.videoPath - 视频路径
   * @param {number} input.videoDuration - 视频时长
   * @param {number} input.fps - 帧率
   * @returns {Promise<Object>} 基础时间轴
   */
  async buildBaseTimeline(input) {
    const { videoPath, videoDuration, fps = 30 } = input;

    this.logger.info('⏱️  TimelineBuilder: 开始构建基础时间轴');
    this.logger.info(`  视频路径: ${videoPath}`);
    this.logger.info(`  视频时长: ${videoDuration}秒`);

    try {
      // 1. 获取带时间戳的语音识别结果
      const asrResult = await this.asrService.transcribeWithTimestamps(videoPath);

      // 2. 分析语音分段
      const speechSegments = this.analyzeSpeechSegments(asrResult);

      // 3. 识别自然断点
      const insertionPoints = this.identifyInsertionPoints(speechSegments);

      // 4. 构建基础时间轴
      const baseTimeline = {
        version: '1.0',
        videoInfo: {
          duration: videoDuration,
          fps: fps,
          resolution: '1080x1920'
        },
        speechSegments: speechSegments,
        insertionPoints: insertionPoints,
        sceneChanges: []  // 可选功能
      };

      this.logger.info('  ✅ 基础时间轴构建完成');
      this.logger.info(`    - 语音分段: ${speechSegments.length}个`);
      this.logger.info(`    - 插入点: ${insertionPoints.length}个`);

      return { baseTimeline };

    } catch (error) {
      this.logger.error('❌ 时间轴构建失败', { error: error.message });
      throw error;
    }
  }

  /**
   * 分析语音分段
   * @param {Array} asrResult - ASR识别结果
   * @returns {Array} 语音分段
   */
  analyzeSpeechSegments(asrResult) {
    const segments = [];
    const pauseThreshold = 0.5;  // 0.5秒以上算停顿

    for (let i = 0; i < asrResult.length; i++) {
      const current = asrResult[i];
      const next = asrResult[i + 1];

      // 添加语音段
      segments.push({
        id: `speech_${i + 1}`,
        startTime: current.startTime,
        endTime: current.endTime,
        text: current.text,
        confidence: current.confidence || 0.9,
        words: current.words || [],
        isPause: false
      });

      // 检查停顿
      if (next && (next.startTime - current.endTime) > pauseThreshold) {
        const pauseDuration = next.startTime - current.endTime;

        segments.push({
          id: `pause_${i + 1}`,
          startTime: current.endTime,
          endTime: next.startTime,
          text: '',
          isPause: true,
          duration: pauseDuration
        });
      }
    }

    return segments;
  }

  /**
   * 识别插入点
   * @param {Array} speechSegments - 语音分段
   * @returns {Array} 插入点
   */
  identifyInsertionPoints(speechSegments) {
    const points = [];
    let pointId = 1;

    for (let i = 0; i < speechSegments.length; i++) {
      const segment = speechSegments[i];

      if (segment.isPause) {
        // 停顿是最佳插入点
        const duration = segment.endTime - segment.startTime;
        const prevSegment = speechSegments[i - 1];
        const nextSegment = speechSegments[i + 1];

        points.push({
          id: `point_${pointId++}`,
          time: segment.startTime,
          type: 'pause',
          duration: duration,
          suitability: duration > 0.8 ? 'high' : 'medium',
          metadata: {
            pauseLength: duration,
            energyLevel: 'low',
            beforeText: prevSegment ? prevSegment.text : '',
            afterText: nextSegment ? nextSegment.text : ''
          }
        });
      }
    }

    return points;
  }

  /**
   * 获取智能体名称
   * @returns {string}
   */
  getName() {
    return this.name;
  }

  /**
   * 获取智能体状态
   * @returns {Object}
   */
  getStatus() {
    return {
      name: this.name,
      ready: true,
      services: {
        asr: !!this.asrService
      }
    };
  }
}

export default TimelineBuilder;
```

```javascript
   * 场景拆解
   */

    const { task_1_2, videoDuration } = input;
    const understanding = task_1_2.understanding;
    
    console.log('  🎬 SceneDesigner: 场景拆解');
    
    const scenes = [];

    // 规则1: 开场必须是原视频（2秒）

      id: 'scene_0',
      type: 'original',

    });

      const content = contents[i];

      if (lastSceneType !== 'original' && consecutiveComplexScenes < 2) {

          id: `scene_${scenes.length}`,
          type: 'original',
          startTime: currentTime,
          endTime: currentTime + 1,
          purpose: '常规过渡',
          priority: 'medium'
        });
        currentTime += 1;
        lastSceneType = 'original';
      }

      if (content.type === 'viewpoint') {
        const duration = this.calculateOptimalDuration(content.text);
        scenes.push({
          id: `scene_${scenes.length}`,
          type: 'video-with-card',

          endTime: currentTime + duration,
          cardText: content.text,
          cardStyle: content.importance === 'high' ? 'blue' : 'yellow',
          purpose: '观点表述',
          priority: content.importance
        });
        currentTime += duration;
        lastSceneType = 'video-with-card';
        consecutiveComplexScenes++;
        
      } else if (content.type === 'explanation') {
        const duration = this.calculateOptimalDuration(content.text, 'explanation');
        scenes.push({
          id: `scene_${scenes.length}`,
          type: 'multi-layer-composition',
          startTime: currentTime,
          endTime: currentTime + duration,
          keyword: content.keyword,
          needMaterial: true,
          needHighlight: true,
          purpose: '观点解释',
          priority: 'high'
        });
        currentTime += duration;
        lastSceneType = 'multi-layer-composition';
        consecutiveComplexScenes++;
      }
      
      // 规则6: 每3个内容后，插入较长过渡
      if ((i + 1) % 3 === 0 && i < contents.length - 1) {
        scenes.push({
          id: `scene_${scenes.length}`,
          type: 'original',
          startTime: currentTime,
          endTime: currentTime + 2.5,
          purpose: '段落过渡',
          priority: 'high'
        });
        currentTime += 2.5;
        lastSceneType = 'original';
        consecutiveComplexScenes = 0;
      }
    }
    
    // 规则7: 结尾必须是原视频
    const endDuration = Math.min(3, videoDuration - currentTime);
    scenes.push({
      id: `scene_${scenes.length}`,
      type: 'original',
      startTime: currentTime,
      endTime: currentTime + endDuration,
      purpose: '结尾',
      priority: 'critical'
    });
    
    // 验证时间轴
    const validation = await this.validator.validate(scenes, videoDuration);
    
    if (!validation.passed) {
      console.log('  ⚠️ 时间轴验证失败，调整中');
      return await this.adjustTimeline(scenes, validation.issues, videoDuration);
    }
    
    // 计算统计信息
    const stats = this.calculateStats(scenes);
    
    console.log(`  ✅ 场景拆解完成:`);
    console.log(`    - 总场景数: ${scenes.length}`);
    console.log(`    - 原视频占比: ${stats.originalRatio}%`);
    console.log(`    - 卡片场景: ${stats.cardScenes}个`);
    console.log(`    - 多层场景: ${stats.multiLayerScenes}个`);
    
    // 验证原视频占比
    if (stats.originalRatio < 25) {
      console.log('  ⚠️ 原视频占比不足25%，增加过渡');
      return await this.addMoreTransitions(scenes, videoDuration);
    }
    
    return {
      scenes: scenes,
      stats: stats
    };
  }
  
  /**
   * 计算最优时长
   */

    const baseTime = type === 'viewpoint' ? 3 : 5;
    const textLength = text.length;
    
    // 根据文字长度调整时长
    if (textLength <= 10) {
      return baseTime;
    } else if (textLength <= 20) {
      return baseTime + 1;
    } else {
      return baseTime + 2;
    }
  }

    let cardScenes = 0;

      } else if (scene.type === 'multi-layer-composition') {
        multiLayerScenes++;
      }
```

#### 8. SmartCropExpert（智能裁剪专家）

**核心职责**

```javascript
/**
 * 智能裁剪专家 - 动态视频裁剪
 *
 * 职责：
 * 1. 检测视频中的重要区域（人脸、文字、关键对象）
 * 2. 计算最佳裁剪框（避开卡片位置）
 * 3. 动态跟踪（人脸移动时调整裁剪框）
 * 4. 生成安全区域地图
 */
class SmartCropExpert {
  constructor() {
    this.name = 'SmartCropExpert';
    this.faceDetector = new FaceDetectionService();
    this.smartCropService = new SmartCropService();
  }

  /**
   * 分析视频并生成裁剪方案
   */
  async analyzeCropStrategy(input) {
    const { videoPath, cardPositions, targetSize } = input;

    console.log('  ✂️ SmartCropExpert: 分析裁剪策略');

    // 1. 采样多个时间点检测人脸
    const sampleTimes = [0, 2, 5, 8, 10, 15, 20];
    const facePositions = [];

    for (const time of sampleTimes) {
      const face = await this.faceDetector.detectAt(videoPath, time);
      if (face) {
        facePositions.push({
          time: time,
          x: face.x,
          y: face.y,
          width: face.width,
          height: face.height
        });
      }
    }

    console.log(`    检测到 ${facePositions.length} 个时间点的人脸`);

    // 2. 判断人脸是否移动
    const isStatic = this.isFaceStatic(facePositions);
    console.log(`    人脸状态: ${isStatic ? '静态' : '动态'}`);

    // 3. 计算安全区域（避开卡片位置）
    const safeZones = this.calculateSafeZones(cardPositions, targetSize);
    console.log(`    安全区域: ${safeZones.length} 个`);

    // 4. 生成裁剪方案
    let cropStrategy;
    if (isStatic) {
      // 静态裁剪：计算一个固定的裁剪框
      cropStrategy = await this.generateStaticCropStrategy(
        facePositions[0],
        safeZones,
        targetSize
      );
    } else {
      // 动态裁剪：为每个时间段计算裁剪框
      cropStrategy = await this.generateDynamicCropStrategy(
        facePositions,
        safeZones,
        targetSize
      );
    }

    console.log('  ✅ 裁剪策略生成完成');

    return {
      strategy: cropStrategy,
      isStatic: isStatic,
      safeZones: safeZones
    };
  }

  /**
   * 判断人脸是否静态
   */
  isFaceStatic(facePositions) {
    if (facePositions.length < 2) return true;

    // 计算人脸中心点的移动距离
    const movements = [];
    for (let i = 1; i < facePositions.length; i++) {
      const prev = facePositions[i - 1];
      const curr = facePositions[i];

      const centerX1 = prev.x + prev.width / 2;
      const centerY1 = prev.y + prev.height / 2;
      const centerX2 = curr.x + curr.width / 2;
      const centerY2 = curr.y + curr.height / 2;

      const distance = Math.sqrt(
        Math.pow(centerX2 - centerX1, 2) +
        Math.pow(centerY2 - centerY1, 2)
      );

      movements.push(distance);
    }

    // 如果平均移动距离小于50像素，认为是静态
    const avgMovement = movements.reduce((a, b) => a + b, 0) / movements.length;
    return avgMovement < 50;
  }

  /**
   * 计算安全区域（避开卡片位置）
   */
  calculateSafeZones(cardPositions, targetSize) {
    const safeZones = [];

    // 将画面分成9宫格
    const gridWidth = targetSize.width / 3;
    const gridHeight = targetSize.height / 3;

    for (let row = 0; row < 3; row++) {
      for (let col = 0; col < 3; col++) {
        const zone = {
          x: col * gridWidth,
          y: row * gridHeight,
          width: gridWidth,
          height: gridHeight,
          occupied: false
        };

        // 检查是否被卡片占用
        for (const card of cardPositions) {
          if (this.isOverlapping(zone, card)) {
            zone.occupied = true;
            break;
          }
        }

        if (!zone.occupied) {
          safeZones.push(zone);
        }
      }
    }

    return safeZones;
  }

  /**
   * 生成静态裁剪策略
   */
  async generateStaticCropStrategy(facePosition, safeZones, targetSize) {
    // 计算包含人脸的最佳裁剪框
    const faceCenterX = facePosition.x + facePosition.width / 2;
    const faceCenterY = facePosition.y + facePosition.height / 2;

    // 裁剪框尺寸（保持目标比例）
    const aspectRatio = targetSize.width / targetSize.height;
    let cropWidth = facePosition.width * 2.5;  // 人脸宽度的2.5倍
    let cropHeight = cropWidth / aspectRatio;

    // 确保裁剪框不超出视频边界
    cropWidth = Math.min(cropWidth, 1920);
    cropHeight = Math.min(cropHeight, 1080);

    // 计算裁剪框位置（人脸居中）
    let cropX = faceCenterX - cropWidth / 2;
    let cropY = faceCenterY - cropHeight / 2;

    // 边界检查
    cropX = Math.max(0, Math.min(cropX, 1920 - cropWidth));
    cropY = Math.max(0, Math.min(cropY, 1080 - cropHeight));

    return {
      type: 'static',
      crop: {
        x: Math.round(cropX),
        y: Math.round(cropY),
        width: Math.round(cropWidth),
        height: Math.round(cropHeight)
      },
      scale: {
        width: targetSize.width,
        height: targetSize.height
      }
    };
  }

  /**
   * 生成动态裁剪策略
   */
  async generateDynamicCropStrategy(facePositions, safeZones, targetSize) {
    const keyframes = [];

    for (const face of facePositions) {
      const crop = await this.generateStaticCropStrategy(
        face,
        safeZones,
        targetSize
      );

      keyframes.push({
        time: face.time,
        crop: crop.crop
      });
    }

    return {
      type: 'dynamic',
      keyframes: keyframes,
      scale: {
        width: targetSize.width,
        height: targetSize.height
      }
    };
  }

  /**
   * 检查两个区域是否重叠
   */
  isOverlapping(zone1, zone2) {
    return !(
      zone1.x + zone1.width < zone2.x ||
      zone2.x + zone2.width < zone1.x ||
      zone1.y + zone1.height < zone2.y ||
      zone2.y + zone2.height < zone1.y
    );
  }
}
```

```javascript
    const availableZones = [];

      }
    }

      if (middleZone) {

/**
```

```javascript
    const scale = strategy.scale;

    const filterComplex = `crop=${crop.width}:${crop.height}:${crop.x}:${crop.y},scale=${scale.width}:${scale.height}`;

    const cmd = `ffmpeg -i "${videoPath}" -vf "${filterComplex}" -c:v libx264 -preset fast -c:a copy "${outputPath}" -y`;

    console.log('✅ 静态裁剪完成');

   * 动态裁剪（使用关键帧）

    await execAsync(cmd);
```

```javascript
      { name: 'lower-center', x: 140, y: 1200 },
      { name: 'lower-right', x: 240, y: 1200 },

      { name: 'bottom-center', x: 140, y: 1450 }
    ];

        x: preset.x,
        y: preset.y,
        width: cardWidth,
        height: cardHeight

          isAvailable = false;
          break;

        positions.push({
          name: preset.name,
          x: preset.x,
          y: preset.y,
          score: this.calculatePositionScore(preset, forbiddenZones)

      }

   * 计算位置分数

    let score = 100;

    // 1. 距离禁止区域越远越好
    for (const zone of forbiddenZones) {
      const distance = this.calculateDistance(position, zone);
      const penalty = Math.max(0, 100 - distance / 10);
      score -= penalty * (zone.priority / 10);

    // 2. 中间位置加分
    if (position.name.includes('center')) {
      score += 20;

    // 3. 中部区域加分（视觉焦点）
    if (position.name.includes('middle') || position.name.includes('upper')) {
      score += 15;
    }

    return score;

   */

      // 检查是否已被使用
      const isUsed = usedPositions.some(used =>
        Math.abs(used.x - position.x) < 50 &&
        Math.abs(used.y - position.y) < 50
      );

      if (!isUsed) {
        return position;
      }
    }

    // 如果所有位置都被占用，返回分数最高的位置（允许重叠）
    return candidatePositions[0] || { x: 140, y: 860, score: 0 };

   * 检查两个区域是否重叠

  isOverlapping(zone1, zone2) {

      zone1.x + zone1.width < zone2.x ||
      zone2.x + zone2.width < zone1.x ||
      zone1.y + zone1.height < zone2.y ||
      zone2.y + zone2.height < zone1.y
    );
  }

  /**
   * 计算两个区域的距离
   */
  calculateDistance(pos1, pos2) {
    const centerX1 = pos1.x + (pos1.width || 0) / 2;
    const centerY1 = pos1.y + (pos1.height || 0) / 2;
    const centerX2 = pos2.x + pos2.width / 2;
    const centerY2 = pos2.y + pos2.height / 2;

    return Math.sqrt(
      Math.pow(centerX2 - centerX1, 2) +
      Math.pow(centerY2 - centerY1, 2)
    );
  }
```

```javascript
/**

import BaiduSpeechService from './BaiduSpeechService.js';

class TimeSyncService {
  constructor() {

  }

  /**
   * 获取关键词的精准时间戳
   * @param {string} audioPath - 音频路径
   * @param {Array} keywords - 关键词列表
   * @returns {Promise<Array>} 时间戳列表
   */
  async getKeywordTimestamps(audioPath, keywords) {
    console.log('⏱️ 获取关键词时间戳...');
    console.log(`  关键词: ${keywords.join(', ')}`);

    try {
      // 1. 调用百度ASR获取带时间戳的识别结果
      const transcriptWithTimestamps = await this.speechService.recognizeWithTimestamps(audioPath);

      if (!transcriptWithTimestamps || !transcriptWithTimestamps.words) {
        console.log('  ⚠️ 未获取到时间戳，使用估算方法');
        return this.estimateTimestamps(keywords, transcriptWithTimestamps.text);
      }

      // 2. 匹配关键词
      const timestamps = [];

      for (const keyword of keywords) {
        const matches = this.findKeywordInWords(keyword, transcriptWithTimestamps.words);

        if (matches.length > 0) {
          timestamps.push({
            keyword: keyword,
            occurrences: matches,
            firstAppearance: matches[0].startTime
          });

          console.log(`  ✅ "${keyword}" 出现 ${matches.length} 次，首次: ${matches[0].startTime}s`);
        } else {
          console.log(`  ⚠️ "${keyword}" 未找到，使用估算`);
          timestamps.push({
            keyword: keyword,
            occurrences: [],
            firstAppearance: null,
            estimated: true
          });
        }
      }

      return timestamps;

    } catch (error) {
      console.error('❌ 时间戳提取失败:', error);
      // 降级到估算方法
      return this.estimateTimestamps(keywords, null);
    }
  }

  /**
   * 在词列表中查找关键词
   */
  findKeywordInWords(keyword, words) {
    const matches = [];
    const keywordChars = keyword.split('');

    for (let i = 0; i <= words.length - keywordChars.length; i++) {
      let isMatch = true;

      for (let j = 0; j < keywordChars.length; j++) {
        if (words[i + j].word !== keywordChars[j]) {
          isMatch = false;
          break;
        }
      }

      if (isMatch) {
        matches.push({
          startTime: words[i].startTime,
          endTime: words[i + keywordChars.length - 1].endTime,
          text: keyword
        });
      }
    }

    return matches;
  }

  /**
   * 估算关键词时间戳（降级方案）
   */
  estimateTimestamps(keywords, fullText) {
    console.log('  使用估算方法计算时间戳');

    const timestamps = [];
    const avgDuration = 3;  // 假设每个关键词平均3秒

    for (let i = 0; i < keywords.length; i++) {
      timestamps.push({
        keyword: keywords[i],
        occurrences: [{
          startTime: i * avgDuration + 2,
          endTime: i * avgDuration + 2 + avgDuration,
          text: keywords[i]
        }],
        firstAppearance: i * avgDuration + 2,
        estimated: true
      });
    }

    return timestamps;
  }

  /**
   * 优化卡片显示时机
   * @param {Array} scenes - 场景列表
   * @param {Array} keywordTimestamps - 关键词时间戳
   * @returns {Array} 优化后的场景列表
   */
  optimizeCardTiming(scenes, keywordTimestamps) {
    console.log('⏱️ 优化卡片显示时机...');

    const optimizedScenes = [];

    for (const scene of scenes) {
      if (scene.type === 'video-with-card' || scene.type === 'multi-layer-composition') {
        // 查找对应的关键词时间戳
        const keyword = scene.keyword || scene.cardText;
        const timestamp = keywordTimestamps.find(ts =>
          ts.keyword === keyword || keyword.includes(ts.keyword)
        );

        if (timestamp && timestamp.firstAppearance !== null) {
          // 调整场景时间：在关键词出现前0.3秒显示卡片
          const adjustedStartTime = Math.max(0, timestamp.firstAppearance - 0.3);
          const duration = scene.endTime - scene.startTime;

          optimizedScenes.push({
            ...scene,
            startTime: adjustedStartTime,
            endTime: adjustedStartTime + duration,
            syncedWithSpeech: true
          });

          console.log(`  ✅ "${keyword}" 时间调整: ${scene.startTime}s -> ${adjustedStartTime}s`);
        } else {
          // 保持原时间
          optimizedScenes.push({
            ...scene,
            syncedWithSpeech: false
          });
        }
      } else {
        // 原视频场景保持不变
        optimizedScenes.push(scene);
      }
    }

    // 重新排序并调整时间轴
    optimizedScenes.sort((a, b) => a.startTime - b.startTime);

    // 检查并修复时间重叠
    for (let i = 1; i < optimizedScenes.length; i++) {
      const prev = optimizedScenes[i - 1];
      const curr = optimizedScenes[i];

      if (curr.startTime < prev.endTime) {
        // 有重叠，调整当前场景的开始时间
        const gap = 0.5;  // 0.5秒间隔
        curr.startTime = prev.endTime + gap;
        curr.endTime = curr.startTime + (curr.endTime - curr.startTime);

        console.log(`  ⚠️ 修复时间重叠: 场景 ${i} 调整到 ${curr.startTime}s`);
      }
    }

    console.log('✅ 时机优化完成');

    return optimizedScenes;
  }

  /**
   * 计算最佳卡片持续时间
   * @param {string} text - 卡片文字
   * @param {number} speechDuration - 语音持续时间
   * @returns {number} 建议的持续时间（秒）
   */
  calculateOptimalDuration(text, speechDuration = null) {
    // 基础时长：根据文字长度
    const baseTime = Math.max(2, text.length * 0.3);

    // 如果有语音时长，使用语音时长
    if (speechDuration) {
      return Math.max(baseTime, speechDuration + 0.5);
    }

    // 限制最大时长
    return Math.min(baseTime, 6);
  }
}
```

```javascript
/**
 * 质量总监 - 新增检查项

class QualityDirector {
  constructor() {
    this.name = 'QualityDirector';
    this.standards = this.loadQualityStandards();
  }

  /**
   * 检查布局质量
   */
  async checkLayoutQuality(input) {
    const { task_3_6 } = input;
    const layouts = task_3_6.layouts;

    console.log('  🔍 QualityDirector: 检查布局质量');

    const issues = [];

    // 1. 检查是否有卡片位置
    if (!layouts || layouts.length === 0) {
      issues.push({
        type: 'no_layouts',
        severity: 'high',
        message: '未生成卡片布局'
      });
    }

    // 2. 检查卡片是否在画面内
    for (const layout of layouts) {
      if (layout.position.x < 0 || layout.position.y < 0 ||
          layout.position.x + layout.size.width > 1080 ||
          layout.position.y + layout.size.height > 1920) {
        issues.push({
          type: 'out_of_bounds',
          severity: 'high',
          cardId: layout.cardId,
          message: '卡片超出画面边界'
        });
      }
    }

    // 3. 检查卡片是否重叠
    for (let i = 0; i < layouts.length; i++) {
      for (let j = i + 1; j < layouts.length; j++) {
        if (this.isOverlapping(layouts[i], layouts[j])) {
          issues.push({
            type: 'overlapping',
            severity: 'medium',
            cards: [layouts[i].cardId, layouts[j].cardId],
            message: '卡片重叠'
          });
        }
      }
    }

    // 4. 检查布局美观度
    const aestheticScore = this.calculateAestheticScore(layouts);
    if (aestheticScore < 70) {
      issues.push({
        type: 'poor_aesthetics',
        severity: 'low',
        score: aestheticScore,
        message: '布局美观度不足'
      });
    }

    if (issues.length > 0) {
      console.log(`  ⚠️ 发现 ${issues.length} 个布局问题`);

      // 高严重度问题直接失败
      const highIssues = issues.filter(i => i.severity === 'high');
      if (highIssues.length > 0) {
        throw new Error('布局质量不合格');
      }
    }

    console.log('  ✅ 布局质量合格');
    return { passed: true, issues: issues };
  }

  /**
   * 检查时间同步准确性
   */
  async checkTimingAccuracy(input) {
    const { task_3_8 } = input;
    const optimizedScenes = task_3_8;

    console.log('  🔍 QualityDirector: 检查时间同步');

    const issues = [];

    // 1. 检查是否有场景
    if (!optimizedScenes || optimizedScenes.length === 0) {
      issues.push({
        type: 'no_scenes',
        severity: 'high',
        message: '未生成场景'
      });
    }

    // 2. 检查时间轴连续性
    for (let i = 1; i < optimizedScenes.length; i++) {
      const prev = optimizedScenes[i - 1];
      const curr = optimizedScenes[i];

      // 检查间隙
      const gap = curr.startTime - prev.endTime;
      if (gap > 1) {
        issues.push({
          type: 'large_gap',
          severity: 'medium',
          gap: gap,
          message: `场景 ${i-1} 和 ${i} 之间间隙过大: ${gap}s`
        });
      }

      // 检查重叠
      if (gap < 0) {
        issues.push({
          type: 'overlap',
          severity: 'high',
          overlap: Math.abs(gap),
          message: `场景 ${i-1} 和 ${i} 时间重叠: ${Math.abs(gap)}s`
        });
      }
    }

    // 3. 检查同步标记
    const syncedScenes = optimizedScenes.filter(s => s.syncedWithSpeech);
    const syncRate = syncedScenes.length / optimizedScenes.length;

    console.log(`  📊 时间同步率: ${(syncRate * 100).toFixed(2)}%`);

    if (syncRate < 0.5) {
      issues.push({
        type: 'low_sync_rate',
        severity: 'medium',
        rate: syncRate,
        message: '时间同步率过低'
      });
    }

    if (issues.length > 0) {
      console.log(`  ⚠️ 发现 ${issues.length} 个时间同步问题`);

      const highIssues = issues.filter(i => i.severity === 'high');
      if (highIssues.length > 0) {
        throw new Error('时间同步质量不合格');
      }
    }

    console.log('  ✅ 时间同步质量合格');
    return { passed: true, issues: issues, syncRate: syncRate };
  }

  /**
   * 计算美观度分数
   */
  calculateAestheticScore(layouts) {
    let score = 100;

    // 1. 对称性检查
    const centerX = 540;  // 1080 / 2
    let symmetryScore = 0;

    for (const layout of layouts) {
      const cardCenterX = layout.position.x + layout.size.width / 2;
      const distanceFromCenter = Math.abs(cardCenterX - centerX);
      symmetryScore += Math.max(0, 100 - distanceFromCenter / 5);
    }

    symmetryScore = symmetryScore / layouts.length;
    score = score * 0.3 + symmetryScore * 0.3;

    // 2. 间距均匀性
    if (layouts.length > 1) {
      const gaps = [];
      for (let i = 1; i < layouts.length; i++) {
        gaps.push(layouts[i].position.y - (layouts[i-1].position.y + layouts[i-1].size.height));
      }

      const avgGap = gaps.reduce((a, b) => a + b, 0) / gaps.length;
      const variance = gaps.reduce((sum, gap) => sum + Math.pow(gap - avgGap, 2), 0) / gaps.length;
      const uniformityScore = Math.max(0, 100 - variance / 10);

      score = score * 0.7 + uniformityScore * 0.3;
    }

    return Math.round(score);
  }

  /**
   * 检查重叠
   */
  isOverlapping(layout1, layout2) {
    const l1 = {
      x: layout1.position.x,
      y: layout1.position.y,
      width: layout1.size.width,
      height: layout1.size.height
    };

    const l2 = {
      x: layout2.position.x,
      y: layout2.position.y,
      width: layout2.size.width,
      height: layout2.size.height
    };

    return !(
      l1.x + l1.width < l2.x ||
      l2.x + l2.width < l1.x ||
      l1.y + l1.height < l2.y ||
      l2.y + l2.height < l1.y
    );
  }
}
```

```javascript
class PerformanceMonitor {
  constructor() {
    this.metrics = {};
  }

  /**
   * 开始计时
   */
  start(taskName) {
    this.metrics[taskName] = {
      startTime: Date.now(),
      endTime: null,
      duration: null
    };
  }

  /**
   * 结束计时
   */
  end(taskName) {
    if (this.metrics[taskName]) {
      this.metrics[taskName].endTime = Date.now();
      this.metrics[taskName].duration = 
        this.metrics[taskName].endTime - this.metrics[taskName].startTime;
    }
  }

  /**
   * 生成报告
   */
  generateReport() {
    console.log('\n⏱️ 性能报告：\n');
    console.log('任务名称'.padEnd(40) + '耗时');
    console.log('─'.repeat(60));

    for (const [taskName, metric] of Object.entries(this.metrics)) {
      if (metric.duration) {
        const duration = (metric.duration / 1000).toFixed(2);
        console.log(taskName.padEnd(40) + `${duration}s`);
      }
    }

    console.log('─'.repeat(60));

    const totalDuration = Object.values(this.metrics)
      .reduce((sum, m) => sum + (m.duration || 0), 0);

    console.log('总耗时'.padEnd(40) + `${(totalDuration / 1000).toFixed(2)}s\n`);
  }
}
```

```javascript
// 三级错误处理
class ErrorHandler {
  /**
   * Level 1: 重试
   */
  async retryOnFailure(fn, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
      try {
        return await fn();
      } catch (error) {
        if (i === maxRetries - 1) throw error;
        console.log(`  ⚠️ 重试 ${i + 1}/${maxRetries}`);
        await this.sleep(1000 * (i + 1));
      }
    }
  }

  /**
   * Level 2: 降级
   */
  async fallbackOnFailure(fn, fallbackFn) {
    try {
      return await fn();
    } catch (error) {
      console.log('  ⚠️ 主方法失败，使用降级方案');
      return await fallbackFn();
    }
  }

  /**
   * Level 3: 跳过
   */
  async skipOnFailure(fn, defaultValue) {
    try {
      return await fn();
    } catch (error) {
      console.log('  ⚠️ 任务失败，使用默认值');
      return defaultValue;
    }
  }
}
```

```javascript
// 多级缓存
class CacheManager {
  constructor() {
    this.memoryCache = new Map();
    this.diskCache = new DiskCache();
  }

  async get(key) {
    // Level 1: 内存缓存
    if (this.memoryCache.has(key)) {
      return this.memoryCache.get(key);
    }

    // Level 2: 磁盘缓存
    const diskValue = await this.diskCache.get(key);
    if (diskValue) {
      this.memoryCache.set(key, diskValue);
      return diskValue;
    }

    return null;
  }

  async set(key, value) {
    // 同时写入两级缓存
    this.memoryCache.set(key, value);
    await this.diskCache.set(key, value);
  }
}
```

```javascript
// src/core/DataManager.js
import fs from 'fs';
import path from 'path';

class DataManager {
  constructor() {
    this.storage = new Map();
    this.locks = new Map();
    this.dataDir = path.join(process.cwd(), 'data');
    
    // 确保数据目录存在
    if (!fs.existsSync(this.dataDir)) {
      fs.mkdirSync(this.dataDir, { recursive: true });
    }
  }
  
  async set(key, value, options = {}) {
    // 获取锁
    await this.acquireLock(key);
    
    try {
      // 存储到内存
      this.storage.set(key, {
        value: value,
        timestamp: Date.now(),
        metadata: options.metadata || {}
      });
      
      // 如果需要持久化
      if (options.persist) {
        const filePath = path.join(this.dataDir, `${key}.json`);
        fs.writeFileSync(filePath, JSON.stringify(value, null, 2));
      }
      
    } finally {
      this.releaseLock(key);
    }
  }
  
  async get(key) {
    const data = this.storage.get(key);
    
    if (!data) {
      // 尝试从文件加载
      const filePath = path.join(this.dataDir, `${key}.json`);
      if (fs.existsSync(filePath)) {
        const content = fs.readFileSync(filePath, 'utf-8');
        return JSON.parse(content);
      }
      return null;
    }
    
    return data.value;
  }
  
  async acquireLock(key, timeout = 30000) {
    const startTime = Date.now();
    
    while (this.locks.get(key)) {
      if (Date.now() - startTime > timeout) {
        throw new Error(`Lock timeout: ${key}`);
      }
      await this.sleep(100);
    }
    
    this.locks.set(key, true);
  }
  
  releaseLock(key) {
    this.locks.delete(key);
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

export default DataManager;
```

```javascript
  }
  
  async speechToText(input) {
    const { videoPath } = input;
    
    console.log('  🎤 ContentAnalyst: 语音识别');
    
    // 提取音频
    const audioPath = await this.extractAudio(videoPath);
    
    // 调用百度ASR
    const transcript = await this.speechService.recognize(audioPath);
    
    if (!transcript || transcript.length < 10) {
      throw new Error('语音识别失败或内容过短');
    }
    
    console.log(`  ✅ 识别到 ${transcript.length} 个字符`);
    
    return {
      transcript: transcript,
      audioPath: audioPath
    };
  }
  
  async extractAudio(videoPath) {
    const { execSync } = require('child_process');
    const audioPath = videoPath.replace('.mp4', '.wav');
    
    const cmd = `ffmpeg -i "${videoPath}" -vn -acodec pcm_s16le -ar 16000 -ac 1 "${audioPath}"`;
    execSync(cmd);
    
    return audioPath;
  }
  
  async analyzeWithWenxin(input) {
    const { task_1_1 } = input;
    const transcript = task_1_1.transcript;
    
    console.log('  🧠 ContentAnalyst: 文心一言分析');
    
    const prompt = this.buildAnalysisPrompt(transcript);
    
    let response = await this.wenxinAPI.chat(prompt, {
      temperature: 0.3
    });
    
    // 解析JSON
    const analysis = this.parseJSON(response);
    
    console.log(`  ✅ 提取到 ${analysis.keywords.length} 个关键词`);
    
    return {
      understanding: analysis,
      transcript: transcript
    };
  }
  
  buildAnalysisPrompt(transcript) {
    return `请分析以下文字稿...`; // 使用前面定义的完整提示词
  }
  
  parseJSON(text) {
    // 提取JSON部分
    const match = text.match(/```json\n([\s\S]*?)\n```/) || 
                  text.match(/\{[\s\S]*\}/);
    
    if (match) {
      return JSON.parse(match[1] || match[0]);
    }
    
    throw new Error('无法解析JSON结果');
  }
}
```

```javascript
class MetricsCollector {
  constructor() {

  }
  
  recordRequest(result) {
    this.metrics.requests.total++;
    
    if (result.success) {
      this.metrics.requests.success++;
    } else {
      this.metrics.requests.failed++;
    }
    
    // 记录性能
    this.recordPerformance(result.duration);
    
    // 记录质量
    if (result.report) {
      this.recordQuality(result.report.score);
    }
    
    // 记录错误
    if (result.error) {
      this.recordError(result.error);
    }
  }
  
  getMetrics() {
    return {
      ...this.metrics,
      successRate: (this.metrics.requests.success / this.metrics.requests.total * 100).toFixed(2) + '%'
    };
  }
}
```

```javascript
class ContentAnalyst {
  async analyzeWithWenxin(input) {
    const { task_0 } = input;  // 获取TimelineBuilder的输出
    const baseTimeline = task_0.baseTimeline;

    // 基于语音分段进行分析
    const transcript = baseTimeline.speechSegments
      .filter(s => !s.isPause)
      .map(s => s.text)
      .join(' ');

    // 调用文心一言分析
    const understanding = await this.qianfanService.analyzeContent(transcript);

    return { understanding };
  }

  // 新增：映射到时间轴
  async mapToTimeline(input) {
    const { task_0, task_1_1 } = input;
    const baseTimeline = task_0.baseTimeline;
    const understanding = task_1_1.understanding;

    // 将观点映射到插入点
    understanding.viewpoints = understanding.viewpoints.map((vp, index) => {
      const point = baseTimeline.insertionPoints[index];
      return {
        ...vp,
        startTime: point ? point.time : 0,
        insertionPoint: point
      };
    });

    return { understanding };
  }
}
```

```javascript
analyzeSpeechSegments(asrResult) {
  const segments = [];

  for (const sentence of asrResult) {
    const words = sentence.text.split('');
    const wordDuration = (sentence.endTime - sentence.startTime) / words.length;

    const wordTimestamps = words.map((word, index) => ({
      word: word,
      startTime: sentence.startTime + index * wordDuration,
      endTime: sentence.startTime + (index + 1) * wordDuration
    }));

    segments.push({
      ...sentence,
      words: wordTimestamps
    });
  }

  return segments;
}
```
