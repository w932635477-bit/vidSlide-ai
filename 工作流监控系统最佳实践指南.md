## 🎯 代码质量最佳实践

### 1. 使用 TypeScript 类型定义

虽然当前使用 JavaScript，但建议添加 JSDoc 类型注释：

```javascript
/**
 * 执行工作流
 * @param {Object} initialData - 初始数据
 * @param {File} initialData.videoFile - 视频文件
 * @returns {Promise<WorkflowResult>} 工作流结果
 * @throws {Error} 当工作流执行失败时
 */
async executeWorkflow(initialData) {
  // ...
}
```

### 2. 使用常量而非魔法数字

```javascript
// ❌ 不好
if (retryCount < 3) {
  // ...
}

// ✅ 好
const MAX_RETRIES = 3
if (retryCount < MAX_RETRIES) {
  // ...
}
```

### 3. 单一职责原则

每个类/函数只做一件事：

```javascript
// ❌ 不好 - 一个函数做太多事
async function processVideo(file) {
  const metadata = extractMetadata(file)
  const transcript = recognizeSpeech(file)
  const keywords = extractKeywords(transcript)
  const template = recommendTemplate(keywords)
  return { metadata, transcript, keywords, template }
}

// ✅ 好 - 分离职责
async function analyzeVideo(file) {
  const metadata = await extractMetadata(file)
  const transcript = await recognizeSpeech(file)
  return { metadata, transcript }
}

async function processContent(transcript) {
  const keywords = await extractKeywords(transcript)
  const template = await recommendTemplate(keywords)
  return { keywords, template }
}
```

### 4. 使用 async/await 而非 Promise 链

```javascript
// ❌ 不好
function processVideo(file) {
  return loadVideo(file)
    .then(video => extractFrames(video))
    .then(frames => analyzeFrames(frames))
    .then(result => result)
}

// ✅ 好
async function processVideo(file) {
  const video = await loadVideo(file)
  const frames = await extractFrames(video)
  const result = await analyzeFrames(frames)
  return result
}
```

### 5. 错误处理要具体

```javascript
// ❌ 不好
catch (error) {
  console.error('Error:', error)
  throw error
}

// ✅ 好
catch (error) {
  if (error.code === 'NETWORK_ERROR') {
    throw new Error('网络连接失败，请检查网络设置')
  } else if (error.code === 'FILE_TOO_LARGE') {
    throw new Error(`文件过大: ${error.size}MB，最大支持 ${MAX_FILE_SIZE}MB`)
  } else {
    throw new Error(`视频处理失败: ${error.message}`)
  }
}
```

## ⚡ 性能优化最佳实践

### 1. 使用对象池减少内存分配

```javascript
class ObjectPool {
  constructor(createFn, resetFn, initialSize = 10) {
    this.createFn = createFn
    this.resetFn = resetFn
    this.pool = []

    // 预分配对象
    for (let i = 0; i < initialSize; i++) {
      this.pool.push(createFn())
    }
  }

  acquire() {
    return this.pool.length > 0
      ? this.pool.pop()
      : this.createFn()
  }

  release(obj) {
    this.resetFn(obj)
    this.pool.push(obj)
  }
}

// 使用示例
const logPool = new ObjectPool(
  () => ({ time: '', level: '', message: '', data: null }),
  (log) => { log.time = ''; log.level = ''; log.message = ''; log.data = null }
)
```

### 2. 批量处理日志

```javascript
// ✅ 使用批量处理
class LogBatcher {
  constructor(batchSize = 10, flushInterval = 100) {
    this.buffer = []
    this.batchSize = batchSize
    this.flushInterval = flushInterval
    this.timer = null
  }

  add(log) {
    this.buffer.push(log)

    if (this.buffer.length >= this.batchSize) {
      this.flush()
    } else if (!this.timer) {
      this.timer = setTimeout(() => this.flush(), this.flushInterval)
    }
  }

  flush() {
    if (this.buffer.length === 0) return

    // 批量处理
    this.processBatch(this.buffer)
    this.buffer = []

    if (this.timer) {
      clearTimeout(this.timer)
      this.timer = null
    }
  }
}
```

### 3. 使用防抖和节流

```javascript
// 防抖 - 延迟执行
function debounce(fn, delay) {
  let timer = null
  return function(...args) {
    clearTimeout(timer)
    timer = setTimeout(() => fn.apply(this, args), delay)
  }
}

// 节流 - 限制频率
function throttle(fn, interval) {
  let lastTime = 0
  return function(...args) {
    const now = Date.now()
    if (now - lastTime >= interval) {
      lastTime = now
      fn.apply(this, args)
    }
  }
}

// 使用示例
const updateUI = debounce(() => {
  workflowSteps.value = agent.getAllSteps()
}, 100)
```

## 🛡️ 错误处理最佳实践

### 1. 分层错误处理

```javascript
// 业务层
class VideoProcessingError extends Error {
  constructor(message, code, details) {
    super(message)
    this.name = 'VideoProcessingError'
    this.code = code
    this.details = details
  }
}

// 使用
try {
  await processVideo(file)
} catch (error) {
  if (error instanceof VideoProcessingError) {
    // 处理业务错误
    handleBusinessError(error)
  } else {
    // 处理系统错误
    handleSystemError(error)
  }
}
```

### 2. 提供有用的错误信息

### 3. 错误恢复策略

```javascript
async function executeWithRetry(fn, maxRetries = 3, delay = 1000) {
  let lastError

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn()
    } catch (error) {
      lastError = error
      console.warn(`尝试 ${i + 1}/${maxRetries} 失败:`, error.message)

      if (i < maxRetries - 1) {
        await new Promise(resolve => setTimeout(resolve, delay * (i + 1)))
      }
    }
  }

  throw lastError
}
```

### 4. 降级处理

```javascript
async function getDataWithFallback() {

    }
  }
}
```

## 🔒 安全最佳实践

### 1. 输入验证

```javascript
function validateVideoFile(file) {
  // 检查文件类型
  const allowedTypes = ['video/mp4', 'video/mov', 'video/avi']
  if (!allowedTypes.includes(file.type)) {
    throw new Error('不支持的文件格式')
  }

  // 检查文件大小
  const maxSize = 500 * 1024 * 1024 // 500MB
  if (file.size > maxSize) {
    throw new Error('文件过大')
  }

  // 检查文件名
  const fileName = file.name
  if (!/^[a-zA-Z0-9_\-\.]+$/.test(fileName)) {
    throw new Error('文件名包含非法字符')
  }

  return true
}
```

### 2. 数据清理

```javascript
// ✅ 清理用户输入
function sanitizeInput(input) {
  if (typeof input === 'string') {
    return input
      .trim()
      .replace(/<script>/gi, '')
```

### 3. 错误信息脱敏
