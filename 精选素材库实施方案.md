## 核心理念

**不存储素材文件，只存储"素材配方"（搜索策略）**

// 快速查找配方
export function findRecipe(id) {
  for (const category of Object.values(materialRecipes)) {
    const recipe = category.find(r => r.id === id)
    if (recipe) return recipe
  }
  return null
}

// 按标签搜索配方
export function searchRecipes(tags) {
  const results = []
  for (const category of Object.values(materialRecipes)) {
    for (const recipe of category) {
      if (tags.some(tag => recipe.tags.includes(tag))) {
        results.push(recipe)
      }
    }
  }
  return results
}

// 获取推荐配方
export function getRecommendedRecipes(context) {
  // 根据上下文推荐配方
  const { type, scene, keywords } = context

  // 简单的推荐逻辑
  let category = 'technology'
  if (keywords.some(k => k.includes('商务') || k.includes('business'))) {
    category = 'business'
  } else if (keywords.some(k => k.includes('数据') || k.includes('data'))) {
    category = 'dataVisualization'
  }

  return materialRecipes[category] || []
}

### 2. 集成到 MaterialService

```javascript
// src/services/MaterialService.js

import { materialRecipes, findRecipe, searchRecipes } from '../data/curatedMaterialRecipes.js'

class MaterialService {
  // ... 现有代码

  /**
   * 获取精选素材（使用配方）
   */
  async getCuratedMaterial(recipeId) {
    const recipe = findRecipe(recipeId)
    if (!recipe) {
      throw new Error(`配方不存在: ${recipeId}`)
    }

    // 1. 先查SmartCache
    const cached = await this.smartCache.search(recipe.searchQuery, {
      limit: 1
    })

    if (cached.length > 0) {
      console.log(`💾 使用缓存的精选素材: ${recipe.name}`)
      return cached[0]
    }

    // 2. 使用配方从外部API获取
    console.log(`🔍 使用配方获取素材: ${recipe.name}`)
    const results = await this.searchMaterials(recipe.searchQuery, {
      limit: 5,
      platforms: recipe.platforms,
      filters: recipe.filters
    })

    if (results.materials.length === 0) {
      throw new Error(`未找到符合配方的素材: ${recipe.name}`)
    }

    // 3. 返回最佳匹配
    return results.materials[0]
  }

  /**
   * 获取推荐的精选素材
   */
  async getRecommendedCuratedMaterials(context, limit = 10) {
    const recipes = getRecommendedRecipes(context)
    const materials = []

    for (const recipe of recipes.slice(0, limit)) {
      try {
        const material = await this.getCuratedMaterial(recipe.id)
        materials.push({
          ...material,
          recipe: recipe,
          isCurated: true
        })
      } catch (error) {
        console.warn(`获取精选素材失败: ${recipe.name}`, error)
      }
    }

    return materials
  }

  /**
   * 浏览精选素材库
   */
  browseCuratedLibrary(category = null) {
    if (category) {
      return materialRecipes[category] || []
    }

    // 返回所有配方
    const allRecipes = []
    for (const [cat, recipes] of Object.entries(materialRecipes)) {
      allRecipes.push(...recipes.map(r => ({ ...r, category: cat })))
    }

    return allRecipes
  }
}
```

## 优势总结

### ✅ 这个方案的优势

1. **零存储成本**
   - 只存储配方（JSON），不存储图片
   - 配方文件 < 50KB

2. **始终最新**
   - 每次都从外部API获取最新素材
   - 可以随时调整配方

3. **智能缓存**
   - 用户使用过的精选素材会自动缓存
   - 下次访问更快

4. **易于维护**
   - 只需要维护配方（搜索策略）
   - 不需要下载、分类、打标签

5. **灵活扩展**
   - 可以随时添加新配方
   - 可以根据用户反馈优化搜索策略

### 📊 对比传统本地素材库

| 特性 | 传统本地库 | 精选配方库 |
|------|-----------|-----------|
| 存储空间 | 500MB-2GB | < 50KB |
| 维护成本 | 高（需要人工筛选） | 低（只需调整配方） |
| 素材新鲜度 | 固定，需要更新 | 始终最新 |
| 素材多样性 | 有限（100-200个） | 无限（API提供） |
| 离线可用 | 是 | 部分（缓存过的） |
| 扩展性 | 困难 | 容易 |

## 总结

**不要建立传统的本地素材库！**

使用"精选配方库"方案：
- ✅ 零存储成本
- ✅ 始终最新
- ✅ 易于维护
- ✅ 配合 SmartCache 实现离线可用

这样既解决了您担心的"下载一堆毫无用处的素材"问题，又提供了精选推荐的价值。
