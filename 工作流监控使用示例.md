# 工作流监控台使用示例

## 基本使用

### 1. 在组件中导入

```vue
<script setup>
import WorkflowMonitor from '@/components/WorkflowMonitor.vue'
import { ref } from 'vue'

// 步骤数据
const steps = ref([
  {
    id: 1,
    name: '视频上传与验证',
    status: 'pending',
    expanded: false,
    input: { fileName: 'video.mp4', size: '50MB' },
    logs: [],
    output: null,
    error: null,
    duration: 0
  },
  {
    id: 2,
    name: '视频格式转换',
    status: 'pending',
    expanded: false,
    input: null,
    logs: [],
    output: null,
    error: null,
    duration: 0
  }
  // ... 更多步骤
])

// 当前步骤索引
const currentStepIndex = ref(0)

// 工作流状态
const isRunning = ref(false)
const isPaused = ref(false)
const isCompleted = ref(false)
const hasError = ref(false)

// 统计信息
const statistics = ref({
  totalSteps: 10,
  completedSteps: 0,
  failedSteps: 0,
  totalDuration: 0
})
</script>

<template>
  <WorkflowMonitor
    :steps="steps"
    :currentStepIndex="currentStepIndex"
    :isRunning="isRunning"
    :isPaused="isPaused"
    :isCompleted="isCompleted"
    :hasError="hasError"
    :statistics="statistics"
    @pause="handlePause"
    @resume="handleResume"
    @cancel="handleCancel"
    @clear-logs="handleClearLogs"
  />
</template>
```

### 2. 步骤状态说明

```javascript
// 步骤状态类型
const stepStatus = {
  pending: '待执行',    // 灰色边框，半透明
  running: '执行中',    // 蓝色边框，脉冲动画
  success: '已完成',    // 绿色边框，淡入动画
  error: '执行失败',    // 红色边框
  retrying: '重试中'    // 黄色边框
}
```

### 3. 执行工作流示例

```javascript
async function executeWorkflow() {
  isRunning.value = true
  
  for (let i = 0; i < steps.value.length; i++) {
    currentStepIndex.value = i
    
    // 更新步骤状态为执行中
    steps.value[i].status = 'running'
    steps.value[i].startTime = Date.now()
    
    try {
      // 执行步骤逻辑
      const result = await executeStep(steps.value[i])
      
      // 更新步骤状态为成功
      steps.value[i].status = 'success'
      steps.value[i].output = result
      steps.value[i].duration = Date.now() - steps.value[i].startTime
      
      // 更新统计信息
      statistics.value.completedSteps++
      statistics.value.totalDuration += steps.value[i].duration
      
    } catch (error) {
      // 更新步骤状态为失败
      steps.value[i].status = 'error'
      steps.value[i].error = {
        message: error.message,
        timestamp: new Date().toISOString(),
        stack: error.stack
      }
      
      statistics.value.failedSteps++
      hasError.value = true
      break
    }
  }
  
  isRunning.value = false
  isCompleted.value = !hasError.value
}
```

### 4. 添加日志

```javascript
function addLog(stepIndex, level, message, data = null) {
  steps.value[stepIndex].logs.push({
    time: new Date().toLocaleTimeString(),
    level: level, // 'info', 'success', 'warning', 'error'
    message: message,
    data: data
  })
}

// 使用示例
addLog(0, 'info', '开始上传视频')
addLog(0, 'success', '视频上传成功', { url: 'https://...' })
addLog(0, 'warning', '视频质量较低')
addLog(0, 'error', '上传失败', { error: 'Network error' })
```

## 自动滚动功能

### 工作原理

1. **监听步骤索引变化**
   - 当 `currentStepIndex` 改变时
   - 自动滚动到新的当前步骤
   - 并展开该步骤的详情

2. **监听步骤状态变化**
   - 当任何步骤的 `status` 改变时
   - 自动滚动到当前步骤
   - 确保用户看到最新状态

3. **滚动配置**
   ```javascript
   element.scrollIntoView({
     behavior: 'smooth',  // 平滑滚动
     block: 'center',     // 居中显示
     inline: 'nearest'    // 最小水平滚动
   })
   ```

### 视觉效果

- **正在执行**：蓝色边框 + 脉冲动画
- **执行成功**：绿色边框 + 淡入动画
- **执行失败**：红色边框
- **待执行**：灰色边框 + 半透明

## 完整示例

```vue
<template>
  <div class="workspace">
    <div class="main-area">
      <!-- 主工作区 -->
    </div>
    
    <div class="right-panel">
      <WorkflowMonitor
        :steps="workflowSteps"
        :currentStepIndex="currentStep"
        :isRunning="isWorkflowRunning"
        :isPaused="isWorkflowPaused"
        :isCompleted="isWorkflowCompleted"
        :hasError="hasWorkflowError"
        :statistics="workflowStats"
        @pause="pauseWorkflow"
        @resume="resumeWorkflow"
        @cancel="cancelWorkflow"
        @clear-logs="clearWorkflowLogs"
      />
    </div>
  </div>
</template>

<script setup>
import { ref, computed } from 'vue'
import WorkflowMonitor from '@/components/WorkflowMonitor.vue'

// 工作流步骤
const workflowSteps = ref([
  {
    id: 1,
    name: '视频上传与验证',
    status: 'pending',
    expanded: false,
    input: null,
    logs: [],
    output: null,
    error: null,
    duration: 0
  },
  // ... 更多步骤
])

// 工作流状态
const currentStep = ref(-1)
const isWorkflowRunning = ref(false)
const isWorkflowPaused = ref(false)
const isWorkflowCompleted = ref(false)
const hasWorkflowError = ref(false)

// 统计信息
const workflowStats = computed(() => ({
  totalSteps: workflowSteps.value.length,
  completedSteps: workflowSteps.value.filter(s => s.status === 'success').length,
  failedSteps: workflowSteps.value.filter(s => s.status === 'error').length,
  totalDuration: workflowSteps.value.reduce((sum, s) => sum + (s.duration || 0), 0)
}))

// 事件处理
const pauseWorkflow = () => {
  isWorkflowRunning.value = false
  isWorkflowPaused.value = true
}

const resumeWorkflow = () => {
  isWorkflowRunning.value = true
  isWorkflowPaused.value = false
}

const cancelWorkflow = () => {
  isWorkflowRunning.value = false
  isWorkflowPaused.value = false
  currentStep.value = -1
}

const clearWorkflowLogs = () => {
  workflowSteps.value.forEach(step => {
    step.logs = []
  })
}

// 执行工作流
async function startWorkflow() {
  isWorkflowRunning.value = true
  hasWorkflowError.value = false
  
  for (let i = 0; i < workflowSteps.value.length; i++) {
    if (!isWorkflowRunning.value) break
    
    currentStep.value = i
    const step = workflowSteps.value[i]
    
    step.status = 'running'
    step.startTime = Date.now()
    
    try {
      // 模拟步骤执行
      await new Promise(resolve => setTimeout(resolve, 2000))
      
      step.status = 'success'
      step.duration = Date.now() - step.startTime
      
    } catch (error) {
      step.status = 'error'
      step.error = {
        message: error.message,
        timestamp: new Date().toISOString(),
        stack: error.stack
      }
      hasWorkflowError.value = true
      break
    }
  }
  
  isWorkflowRunning.value = false
  isWorkflowCompleted.value = !hasWorkflowError.value
}
</script>

<style scoped>
.workspace {
  display: flex;
  height: 100vh;
}

.main-area {
  flex: 1;
}

.right-panel {
  width: 400px;
  border-left: 1px solid #333;
}
</style>
```
